cmake_minimum_required(VERSION 3.8)
project(ares_test_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(USB REQUIRED IMPORTED_TARGET libusb-1.0)
find_package(Threads REQUIRED)
add_executable(chassis_bridge src/chassis_bridge.cpp vendor/ares_protocol.cpp)
target_include_directories(chassis_bridge PRIVATE vendor)
target_link_libraries(chassis_bridge PkgConfig::USB Threads::Threads)
ament_target_dependencies(chassis_bridge rclcpp geometry_msgs std_msgs)
install(TARGETS chassis_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
